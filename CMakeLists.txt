option(DJ_ENABLE_AVX512VNNI "Enable AVX-512 VNNI for int8 L2 distance kernels" ON)
if(DJ_ENABLE_AVX512VNNI)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-mavx512f -mavx512bw -mavx512vnni" HAS_AVX512VNNI_FLAGS)
    if(NOT HAS_AVX512VNNI_FLAGS)
        message(FATAL_ERROR "DJ_ENABLE_AVX512VNNI=ON but compiler does not support -mavx512vnni")
    endif()
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mavx512f -mavx512bw -mavx512vnni")
endif()

endif()
option(DJ_USE_OPENBLAS "Route Eigen GEMM through OpenBLAS via EIGEN_USE_BLAS" ON)
if(DJ_USE_OPENBLAS)
    find_package(OpenBLAS QUIET)
    if(OpenBLAS_FOUND)
        if(NOT OPENBLAS_LIBRARIES AND OpenBLAS_LIBRARIES)
            set(OPENBLAS_LIBRARIES ${OpenBLAS_LIBRARIES})
        endif()
        if(NOT OPENBLAS_LIBRARIES AND OpenBLAS_LIBRARY)
            set(OPENBLAS_LIBRARIES ${OpenBLAS_LIBRARY})
        endif()
        if(NOT OPENBLAS_INCLUDE_DIRS AND OpenBLAS_INCLUDE_DIRS)
            set(OPENBLAS_INCLUDE_DIRS ${OpenBLAS_INCLUDE_DIRS})
        endif()
        if(NOT OPENBLAS_INCLUDE_DIRS AND OpenBLAS_INCLUDE_DIR)
            set(OPENBLAS_INCLUDE_DIRS ${OpenBLAS_INCLUDE_DIR})
        endif()
        if(NOT OPENBLAS_LIBRARIES OR NOT OPENBLAS_INCLUDE_DIRS)
            set(OpenBLAS_FOUND FALSE)
        endif()
    endif()
    if(NOT OpenBLAS_FOUND)
        find_package(PkgConfig QUIET)
        if(PKG_CONFIG_FOUND)
            pkg_check_modules(OPENBLAS_PC QUIET openblas)
        endif()
        if(OPENBLAS_PC_FOUND)
            find_library(OPENBLAS_LIBRARY_FROM_PC NAMES openblas HINTS ${OPENBLAS_PC_LIBRARY_DIRS})
            if(OPENBLAS_LIBRARY_FROM_PC)
                set(OPENBLAS_LIBRARIES ${OPENBLAS_LIBRARY_FROM_PC})
            elseif(OPENBLAS_PC_LINK_LIBRARIES)
                set(OPENBLAS_LIBRARIES ${OPENBLAS_PC_LINK_LIBRARIES})
            elseif(OPENBLAS_PC_LDFLAGS)
                set(OPENBLAS_LIBRARIES ${OPENBLAS_PC_LDFLAGS})
            else()
                set(OPENBLAS_LIBRARIES ${OPENBLAS_PC_LIBRARIES})
            endif()
            set(OPENBLAS_INCLUDE_DIRS ${OPENBLAS_PC_INCLUDE_DIRS})
        else()
            find_library(OPENBLAS_LIBRARY NAMES openblas)
            find_path(OPENBLAS_INCLUDE_DIR NAMES cblas.h openblas/cblas.h)
            if(NOT OPENBLAS_LIBRARY OR NOT OPENBLAS_INCLUDE_DIR)
                message(FATAL_ERROR "DJ_USE_OPENBLAS=ON but OpenBLAS not found: install libopenblas-dev or set OPENBLAS_LIBRARY / OPENBLAS_INCLUDE_DIR")
            endif()
            set(OPENBLAS_LIBRARIES ${OPENBLAS_LIBRARY})
            set(OPENBLAS_INCLUDE_DIRS ${OPENBLAS_INCLUDE_DIR})
        endif()
    endif()
    add_definitions(-DEIGEN_USE_BLAS -DDJ_USE_OPENBLAS)
    include_directories(${OPENBLAS_INCLUDE_DIRS})
endif()
